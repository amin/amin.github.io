% The side margins set the measure and are the one dimension not up for
% negotiation --- widening them past 1.9cm pushes the line length beyond what
% 11pt reads comfortably at. Top and bottom are the give: bullets under the
% experience entries cost about six lines, and the bottom margin gave up 0.25cm
% of them without touching a word. The top stays wider than the bottom: it is
% the margin a reader sees first, above a 26pt name that needs the air, and a
% CV that starts tight against the paper edge reads as one that ran out of room.

% A CV is skimmed, not read, and a word split across a line break
% ("represen-/tative") is a stumble in the paragraph that has the least time to
% land. Refusing to hyphenate costs slightly looser interword spacing instead,
% which this measure (~17cm, around 90 characters) has the slack to absorb.
% \exhyphenpenalty covers the hyphens already in the text, so "front-end" and
% "El-Rifai" stay whole too, and \emergencystretch gives TeX a last resort so a
% stubborn line loosens rather than running into the margin.
\hyphenpenalty=10000
\exhyphenpenalty=10000
\emergencystretch=2em

% Three levels of hierarchy, and only three: section (teal caps over a rule),
% role (the only bold in an entry), and supporting detail (grey). Anything
% that needs to stand out borrows colour or letterspacing rather than size,
% which is what keeps the whole thing on one page.
% ---------------------------------------------------------------------------

\newcommand{\sep}{\enspace\textperiodcentered\enspace}

% Gap between entries inside a section. Defined once so the rhythm cannot
% drift the way a dozen hand-tuned \vspace values did.
\newcommand{\entrygap}{\vspace{0.40cm}}

% Bullets under an entry, built on the \list primitive rather than itemize:
% itemize inherits \topsep and \parsep tuned for essays, which opens a hole
% between the lead line and the first bullet that this page has no room for.
% The marker is small and grey because it only has to mark where a line begins
% --- at six of them down one section, a full-size ink dot starts reading as
% content. An 8pt bullet sits on the 11pt baseline low enough to look like a
% stray full stop, so \raisebox lifts it back to the middle of the x-height.
% \leftmargin is set so wrapped lines align under the words rather than under
% the dot, which is what gives the block a left edge to skim.
\newenvironment{cvbullets}{%
  \begin{list}{{\fontsize{8}{8}\selectfont\color{inkmeta}\raisebox{0.55pt}{\textbullet}}}{%
    \setlength{\labelwidth}{0.55em}%
    \setlength{\labelsep}{0.45em}%
    \setlength{\leftmargin}{1.00em}%
    \setlength{\rightmargin}{0pt}%
    \setlength{\itemindent}{0pt}%
    \setlength{\listparindent}{0pt}%
    \setlength{\topsep}{0.10cm}%
    \setlength{\partopsep}{0pt}%
    \setlength{\parsep}{0pt}%
    \setlength{\itemsep}{0.07cm}%
    \fontsize{11}{14.4}\selectfont
  }%
}{%
  \end{list}%
}

% \nointerlineskip after each \par suppresses the baseline glue LaTeX would
% otherwise add, which is what lets the rule sit tight under its label instead
% of drifting halfway to the content below it.
\newcommand{\cvsection}[1]{%
  \par\vspace{0.62cm}%
  % Keep tracking low. Above roughly 100, pdftotext (and CV-screening tools
  % that use it) starts reading the gaps as spaces and extracts "E D U C AT I
  % O N", which destroys keyword matching. 120 was verified to break; 80 and
  % below were verified clean, so 60 leaves margin for stricter parsers.
  {\fontsize{11.5}{13}\selectfont\bfseries\color{accentteal}%
    \textls[60]{\MakeUppercase{#1}}}%
  \par\nointerlineskip\vspace{0.13cm}%
  {\color{rulegrey}\rule{\textwidth}{0.5pt}}%
  \par\nointerlineskip\vspace{0.28cm}%
}

% Role and dates share a baseline; the organisation sits under the role in
% grey, so the eye reads role first and never has to weigh two bolds.
\newcommand{\cventry}[3]{%
  \noindent\begin{tabular*}{\textwidth}{@{\extracolsep{\fill}}lr@{}}
    {\fontsize{12.5}{15}\selectfont\bfseries #1} & \detail{#3}
  \end{tabular*}%
  \par\vspace{0.06cm}%
  \detail{#2}%
  \par\vspace{0.14cm}%
}

\newcommand{\cvproject}[4]{%
  {\fontsize{12.5}{15}\selectfont\bfseries #1}%
  \par\vspace{0.06cm}%
  {\fontsize{10.5}{13.5}\selectfont\link{#2}{#3}\color{inkmeta}\sep #4}%
  \par\vspace{0.14cm}%
}
