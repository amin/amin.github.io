% Gap between entries inside a section. Defined once so the rhythm cannot
% drift the way a dozen hand-tuned \vspace values did.
\newcommand{\entrygap}{\vspace{0.52cm}}

% \nointerlineskip after each \par suppresses the baseline glue LaTeX would
% otherwise add, which is what lets the rule sit tight under its label instead
% of drifting halfway to the content below it.
\newcommand{\cvsection}[1]{%
  \par\vspace{0.74cm}%
  % Keep tracking low. Above roughly 100, pdftotext (and CV-screening tools
  % that use it) starts reading the gaps as spaces and extracts "E D U C AT I
  % O N", which destroys keyword matching. 120 was verified to break; 80 and
  % below were verified clean, so 60 leaves margin for stricter parsers.
  {\fontsize{11.5}{13}\selectfont\bfseries\color{accentteal}%
    \textls[60]{\MakeUppercase{#1}}}%
  \par\nointerlineskip\vspace{0.13cm}%
  {\color{rulegrey}\rule{\textwidth}{0.5pt}}%
  \par\nointerlineskip\vspace{0.34cm}%
}

% Role and dates share a baseline; the organisation sits under the role in
% grey, so the eye reads role first and never has to weigh two bolds.
\newcommand{\cventry}[3]{%
  \noindent\begin{tabular*}{\textwidth}{@{\extracolsep{\fill}}lr@{}}
    {\fontsize{12.5}{15}\selectfont\bfseries #1} & \detail{#3}
  \end{tabular*}%
  \par\vspace{0.06cm}%
  \detail{#2}%
  \par\vspace{0.14cm}%
}

\newcommand{\cvproject}[4]{%
  {\fontsize{12.5}{15}\selectfont\bfseries #1}%
  \par\vspace{0.06cm}%
  {\fontsize{10.5}{13.5}\selectfont\link{#2}{#3}\color{inkmeta}\sep #4}%
  \par\vspace{0.14cm}%
}
